\subsection*{Álgebra}
\begin{itemize}
\item{La diferencia de dos números es $40$ y están en relación de $3$ a $2$. Hallar dichos números.
\begin{tasks}(5)
\task{40 y 80}\task{60 y 100}\task{70 y 110}\task{80 y 120}
\end{tasks}
}
\item{Dos números están en la relación de $2$ a $5$, pero agregando $175$ a uno de ellos y $115$ al otro, ambos resultados son iguales. Hallar el número mayor.
\begin{tasks}(5)
\task{80}\task{90}\task{100}\task{120}\task{130}
\end{tasks}
}
\item{La suma de $2$ números es $270$ y cuando se le agrega $65$ a cada uno de ellos su razón es $3/5$. Hallar el número menor.
\begin{tasks}(5)
\task{75}\task{85}\task{90}\task{95}\task{105}
\end{tasks}
}
\item{En una proporción geométrica continua, la suma de los extremos es $34$ y la diferencia de los mismos es $16$. Hallar la media proporcional.
\begin{tasks}(5)
\task{12}\task{13}\task{15}\task{18}\task{21}
\end{tasks}
}
\item{Si \quad$\dfrac{a}{b} = \dfrac{c}{d} = k$;\quad $a + c = 4$ y además : $\sqrt{ab\,} + \sqrt{cd\;} = 20$;\quad Hallar $k$.
\begin{tasks}(5)
\task{$1/25$}\task{$1/4$}\task{$4$}\task{$20$}\task{$25$}
\end{tasks}
}
\end{itemize}
