\subsection*{Álgebra}
\begin{itemize}
\item{Tres números están entre sí como $4$; $7$ y $11$ y la suma del menor con el mayor de dichos números es $105$. Determinar el menor de estos números.
\begin{tasks}(5)
\task{14}\task{24}\task{28}\task{30}\task{49}
\end{tasks}
}
\item{Tres números están en la misma relación que $5$; $9$ y $13$. Si la suma de ellos es $216$. Indica el mayor de ellos.
\begin{tasks}(5)
\task{88}\task{96}\task{104}\task{108}\task{1002}
\end{tasks}
}
\item{Si \quad$\dfrac{a}{b} = \dfrac{7}{11}$;\quad además \quad$a \cdot b = 308$. Calcular $b - a$. 
\begin{tasks}(5)
\task{2}\task{3}\task{5}\task{7}\task{8}
\end{tasks}
}
\item{Si \quad$\dfrac{a + 5}{b - 5} = \dfrac{2}{3}$;\quad además \quad$a + b = 50$. Calcular $3a - b$.
\begin{tasks}(5)
\task{7}\task{8}\task{9}\task{10}\task{11}
\end{tasks}
}
\item{Si \quad$\dfrac{a}{b} = \dfrac{8}{3}$;\quad $\dfrac{b}{c} = \dfrac{12}{7}$;\quad además \quad$a - c = 50$. Calcular el valor de $b$. 
\begin{tasks}(5)
\task{16}\task{24}\task{25}\task{29}\task{52}
\end{tasks}
}
\end{itemize}
